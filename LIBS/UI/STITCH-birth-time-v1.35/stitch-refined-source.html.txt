    (function() {
      // Confirmed time state: null until user clicks 'Set time'

      let draft = {

        minute: null,
        ampm: null
      };

      // Raw typed input strings retained across mode switches (no stale fallback overwrite)
      let rawTyped = {
        hour: '',
        minute: ''
      };

      // DOM Elements
      const timeSlotTrigger = document.getElementById('timeSlotTrigger');
      const slotDisplay = document.getElementById('slotDisplay');
      const slotActionText = document.getElementById('slotActionText');
      const editorPanel = document.getElementById('birthTimeEditor');

      const modeBtnClock = document.getElementById('modeBtnClock');
      const modeBtnType = document.getElementById('modeBtnType');
      const viewClock = document.getElementById('viewClock');
      const viewType = document.getElementById('viewType');

      const tabDialHour = document.getElementById('tabDialHour');
      const tabDialMinute = document.getElementById('tabDialMinute');
      const clockFaceContainer = document.getElementById('clockFaceContainer');
      const clockHandGroup = document.getElementById('clockHandGroup');
      const dialButtonsContainer = document.getElementById('dialButtonsContainer');

      const clockSelectedHour = document.getElementById('clockSelectedHour');
      const clockSelectedMinute = document.getElementById('clockSelectedMinute');
      const btnMinusMinute = document.getElementById('btnMinusMinute');
      const btnPlusMinute = document.getElementById('btnPlusMinute');

      const btnAM = document.getElementById('btnAM');
      const btnPM = document.getElementById('btnPM');

      const draftTimeReadout = document.getElementById('draftTimeReadout');

      const inputHour = document.getElementById('inputHour');
      const inputMinute = document.getElementById('inputMinute');
      const typeValidationMsg = document.getElementById('typeValidationMsg');

      const btnCancelDraft = document.getElementById('btnCancelDraft');
      const btnSetTime = document.getElementById('btnSetTime');
      const chkUnknown = document.getElementById('chkUnknown');
      const btnBegin = document.getElementById('btnBegin');

      const confirmedOutcomeCard = document.getElementById('confirmedOutcomeCard');
      const outcomeDetailText = document.getElementById('outcomeDetailText');

      // Clear confirmed outcome helper
      function clearOutcome() {
        confirmedOutcomeCard.classList.remove('visible');
        outcomeDetailText.textContent = '';
      }

      // Check if draft is fully valid: hour 1..12, minute 0..59, and explicit AM/PM
      function isDraftValid() {
        const hValid = draft.hour12 !== null && draft.hour12 >= 1 && draft.hour12 <= 12;
        const mValid = draft.minute !== null && draft.minute >= 0 && draft.minute <= 59;
        const ampmValid = draft.ampm === 'AM' || draft.ampm === 'PM';
        return hValid && mValid && ampmValid;
      }

      function formatDraftDisplay() {
        const hh = draft.hour12 !== null ? String(draft.hour12).padStart(2, '0') : '--';
        const mm = draft.minute !== null ? String(draft.minute).padStart(2, '0') : '--';
        const ap = draft.ampm !== null ? draft.ampm : '--';
        return `${hh}:${mm} ${ap}`;
      }

      function updateDraftDisplay() {
        draftTimeReadout.textContent = formatDraftDisplay();
        clockSelectedHour.textContent = draft.hour12 !== null ? draft.hour12 : '--';
        clockSelectedMinute.textContent = draft.minute !== null ? String(draft.minute).padStart(2, '0') : '--';

        // AM/PM state
        if (draft.ampm === 'AM') {
          btnAM.classList.add('active');
          btnAM.setAttribute('aria-pressed', 'true');
          btnPM.classList.remove('active');
          btnPM.setAttribute('aria-pressed', 'false');
        } else if (draft.ampm === 'PM') {
          btnPM.classList.add('active');
          btnPM.setAttribute('aria-pressed', 'true');
          btnAM.classList.remove('active');
          btnAM.setAttribute('aria-pressed', 'false');
        } else {
          btnAM.classList.remove('active');
          btnAM.setAttribute('aria-pressed', 'false');
          btnPM.classList.remove('active');
          btnPM.setAttribute('aria-pressed', 'false');
        }

        renderDial();
        updateSetTimeState();
      }

      function updateSetTimeState() {
        if (activeMode === 'type') {
          validateTypeInputs();
        } else {
          btnSetTime.disabled = !isDraftValid();
        }
      }

      // Render the circular clock dial face
      // Center (135, 135), Radius = 95px
      function renderDial() {
        dialButtonsContainer.innerHTML = '';
        const cx = 135;
        const cy = 135;
        const r = 95;

        let selectedVal = null;
        let handAngle = null;

        if (dialSubmode === 'hours') {
          selectedVal = draft.hour12;
          if (selectedVal !== null) {
            handAngle = (selectedVal % 12) * 30; // 12 -> 0 deg, 1 -> 30 deg, etc.
          }

          // 12 hours: 1 to 12
          for (let i = 1; i <= 12; i++) {
            const angleDeg = (i % 12) * 30;
            const angleRad = (angleDeg - 90) * (Math.PI / 180);
            const x = cx + r * Math.cos(angleRad);
            const y = cy + r * Math.sin(angleRad);

            const btn = document.createElement('button');
            btn.type = 'button';
            btn.className = 'dial-button' + (selectedVal === i ? ' selected' : '');
            btn.style.left = `${x}px`;
            btn.style.top = `${y}px`;
            btn.textContent = String(i);
            btn.setAttribute('role', 'radio');
            btn.setAttribute('aria-checked', selectedVal === i ? 'true' : 'false');
            btn.setAttribute('aria-label', `${i} o'clock`);
            btn.setAttribute('tabindex', selectedVal === i || (selectedVal === null && i === 12) ? '0' : '-1');

            btn.addEventListener('click', () => {
              draft.hour12 = i;
              rawTyped.hour = String(i);
              inputHour.value = String(i);
              // Auto-advance to minutes if minute not yet selected
              if (draft.minute === null) {
                setDialSubmode('minutes');
              } else {
                updateDraftDisplay();
              }
            });

            btn.addEventListener('keydown', handleDialArrowNav);
            dialButtonsContainer.appendChild(btn);
          }
        } else {
          // Minutes submode: 00, 05, 10, ... 55
          selectedVal = draft.minute;
          if (selectedVal !== null) {
            handAngle = (selectedVal / 60) * 360;
          }

          for (let i = 0; i < 12; i++) {
            const minVal = i * 5;
            const angleDeg = (minVal / 60) * 360;
            const angleRad = (angleDeg - 90) * (Math.PI / 180);
            const x = cx + r * Math.cos(angleRad);
            const y = cy + r * Math.sin(angleRad);

            const isSelected = selectedVal === minVal;
            const btn = document.createElement('button');
            btn.type = 'button';
            btn.className = 'dial-button' + (isSelected ? ' selected' : '');
            btn.style.left = `${x}px`;
            btn.style.top = `${y}px`;
            const textVal = String(minVal).padStart(2, '0');
            btn.textContent = textVal;
            btn.setAttribute('role', 'radio');
            btn.setAttribute('aria-checked', isSelected ? 'true' : 'false');
            btn.setAttribute('aria-label', `${textVal} minutes`);
            btn.setAttribute('tabindex', isSelected || (selectedVal === null && minVal === 0) ? '0' : '-1');

            btn.addEventListener('click', () => {
              draft.minute = minVal;
              rawTyped.minute = textVal;
              inputMinute.value = textVal;
              updateDraftDisplay();
            });

            btn.addEventListener('keydown', handleDialArrowNav);
            dialButtonsContainer.appendChild(btn);
          }
        }

        // Rotate Hand SVG
        if (handAngle !== null) {
          clockHandGroup.classList.remove('no-hand');
          clockHandGroup.style.transform = `rotate(${handAngle}deg)`;
        } else {
          clockHandGroup.classList.add('no-hand');
        }
      }

      // Keyboard arrow navigation on circular dial buttons
      function handleDialArrowNav(e) {
        const buttons = Array.from(dialButtonsContainer.querySelectorAll('.dial-button'));
        const index = buttons.indexOf(e.currentTarget);
        if (index === -1) return;

        let nextIndex = null;
        if (e.key === 'ArrowRight' || e.key === 'ArrowDown') {
          e.preventDefault();
          nextIndex = (index + 1) % buttons.length;
        } else if (e.key === 'ArrowLeft' || e.key === 'ArrowUp') {
          e.preventDefault();
          nextIndex = (index - 1 + buttons.length) % buttons.length;
        }

        if (nextIndex !== null) {
          buttons.forEach(b => b.setAttribute('tabindex', '-1'));
          buttons[nextIndex].setAttribute('tabindex', '0');
          buttons[nextIndex].focus();
        }
      }

      // Dial submode switch (Hour dial vs Minute dial)
      function setDialSubmode(submode) {
        dialSubmode = submode;
        if (submode === 'hours') {
          tabDialHour.classList.add('active');
          tabDialHour.setAttribute('aria-selected', 'true');
          tabDialHour.setAttribute('tabindex', '0');
          tabDialMinute.classList.remove('active');
          tabDialMinute.setAttribute('aria-selected', 'false');
          tabDialMinute.setAttribute('tabindex', '-1');
        } else {
          tabDialMinute.classList.add('active');
          tabDialMinute.setAttribute('aria-selected', 'true');
          tabDialMinute.setAttribute('tabindex', '0');
          tabDialHour.classList.remove('active');
          tabDialHour.setAttribute('aria-selected', 'false');
          tabDialHour.setAttribute('tabindex', '-1');
        }
        updateDraftDisplay();
      }

      tabDialHour.addEventListener('click', () => setDialSubmode('hours'));
      tabDialMinute.addEventListener('click', () => setDialSubmode('minutes'));

      // Arrow navigation between submode tabs
      tabDialHour.addEventListener('keydown', (e) => {
        if (e.key === 'ArrowRight') {
          e.preventDefault();
          setDialSubmode('minutes');
          tabDialMinute.focus();
        }
      });
      tabDialMinute.addEventListener('keydown', (e) => {
        if (e.key === 'ArrowLeft') {
          e.preventDefault();
          setDialSubmode('hours');
          tabDialHour.focus();
        }
      });

      // Stepper for exact minute (+1m / -1m)
      btnMinusMinute.addEventListener('click', () => {
        const cur = draft.minute !== null ? draft.minute : 0;
        draft.minute = (cur - 1 + 60) % 60;
        rawTyped.minute = String(draft.minute).padStart(2, '0');
        inputMinute.value = rawTyped.minute;
        setDialSubmode('minutes');
      });

      btnPlusMinute.addEventListener('click', () => {
        const cur = draft.minute !== null ? draft.minute : 0;
        draft.minute = (cur + 1) % 60;
        rawTyped.minute = String(draft.minute).padStart(2, '0');
        inputMinute.value = rawTyped.minute;
        setDialSubmode('minutes');
      });

      // AM/PM buttons
      btnAM.addEventListener('click', () => {
        draft.ampm = 'AM';
        clearOutcome();
        updateDraftDisplay();
      });

      btnPM.addEventListener('click', () => {
        draft.ampm = 'PM';
        clearOutcome();
        updateDraftDisplay();
      });

      // Mode switching (Clock vs Type)
      function setMode(mode, shouldFocusInput = false) {
        activeMode = mode;
        if (mode === 'clock') {
          modeBtnClock.classList.add('active');
          modeBtnClock.setAttribute('aria-selected', 'true');
          modeBtnClock.setAttribute('tabindex', '0');
          modeBtnType.classList.remove('active');
          modeBtnType.setAttribute('aria-selected', 'false');
          modeBtnType.setAttribute('tabindex', '-1');
          viewClock.style.display = 'block';
          viewType.style.display = 'none';
          updateDraftDisplay();
        } else {
          modeBtnType.classList.add('active');
          modeBtnType.setAttribute('aria-selected', 'true');
          modeBtnType.setAttribute('tabindex', '0');
          modeBtnClock.classList.remove('active');
          modeBtnClock.setAttribute('aria-selected', 'false');
          modeBtnClock.setAttribute('tabindex', '-1');
          viewClock.style.display = 'none';
          viewType.style.display = 'flex';

          // Preserve raw typed state
          inputHour.value = rawTyped.hour;
          inputMinute.value = rawTyped.minute;

          validateTypeInputs();

          // Synchronous focus to summon software keyboard directly
          if (shouldFocusInput) {
            inputHour.focus();
            if (rawTyped.hour) {
              inputHour.select();
            }
          }
        }
      }

      modeBtnClock.addEventListener('click', () => setMode('clock', false));
      modeBtnType.addEventListener('click', () => setMode('type', true));

      // Arrow navigation on Mode Tabs
      modeBtnClock.addEventListener('keydown', (e) => {
        if (e.key === 'ArrowRight') {
          e.preventDefault();
          setMode('type', true);
          modeBtnType.focus();
        }
      });
      modeBtnType.addEventListener('keydown', (e) => {
        if (e.key === 'ArrowLeft') {
          e.preventDefault();
          setMode('clock', false);
          modeBtnClock.focus();
        }
      });

      // Validate typed inputs: strictly 1–12, 00–59, and explicit AM/PM
      function validateTypeInputs() {
        typeValidationMsg.textContent = '';
        inputHour.classList.remove('invalid');
        inputMinute.classList.remove('invalid');

        const hStr = inputHour.value.trim();
        const mStr = inputMinute.value.trim();

        let hVal = null;
        let mVal = null;
        let valid = true;

        if (hStr !== '') {
          const parsedH = parseInt(hStr, 10);
          if (isNaN(parsedH) || parsedH < 1 || parsedH > 12) {
            inputHour.classList.add('invalid');
            typeValidationMsg.textContent = 'Hour must be between 1 and 12.';
            valid = false;
          } else {
            hVal = parsedH;
          }
        } else {
          valid = false;
        }

        if (mStr !== '') {
          const parsedM = parseInt(mStr, 10);
          if (isNaN(parsedM) || parsedM < 0 || parsedM > 59) {
            inputMinute.classList.add('invalid');
            if (!typeValidationMsg.textContent) {
              typeValidationMsg.textContent = 'Minute must be between 00 and 59.';
            }
            valid = false;
          } else {
            mVal = parsedM;
          }
        } else {
          valid = false;
        }

        if (!draft.ampm && valid) {
          typeValidationMsg.textContent = 'Select AM or PM above.';
          valid = false;
        }

        // Update draft representation without stale valid fallbacks
        draft.hour12 = hVal;
        draft.minute = mVal;
        draftTimeReadout.textContent = formatDraftDisplay();

        btnSetTime.disabled = !(valid && draft.ampm);
      }

        validateTypeInputs();

      inputMinute.addEventListener('input', () => {
        inputMinute.value = inputMinute.value.replace(/[^0-9]/g, '');
        rawTyped.minute = inputMinute.value;
        clearOutcome();
        validateTypeInputs();
      });

      // Open / Close Editor
      function openEditor() {
        if (isUnknown) return;
        isEditorOpen = true;
        editorPanel.classList.add('open');
        timeSlotTrigger.setAttribute('aria-expanded', 'true');
        slotActionText.textContent = 'Editing…';

        clearOutcome();
        btnBegin.disabled = true; // Disabled during editing

        // If previously confirmed, pre-populate draft; else start fresh empty
        if (confirmedTime) {
          draft = {
            hour12: confirmedTime.hour12,
            minute: confirmedTime.minute,
            ampm: confirmedTime.ampm
          };
          rawTyped.hour = String(confirmedTime.hour12);
          rawTyped.minute = String(confirmedTime.minute).padStart(2, '0');
        } else {
          draft = {
            hour12: null,
            minute: null,
            ampm: null
          };
          rawTyped.hour = '';
          rawTyped.minute = '';
        }

        inputHour.value = rawTyped.hour;
        inputMinute.value = rawTyped.minute;

        setDialSubmode(draft.hour12 === null ? 'hours' : 'minutes');
        setMode('clock', false);
        updateDraftDisplay();

        editorPanel.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
      }

      function closeEditor(restoreFocus = true) {
        isEditorOpen = false;
        editorPanel.classList.remove('open');
        timeSlotTrigger.setAttribute('aria-expanded', 'false');
        slotActionText.textContent = confirmedTime ? 'Change →' : 'Select →';
        updateBeginButtonState();

        if (restoreFocus) {
          timeSlotTrigger.focus();
        }
      }

      timeSlotTrigger.addEventListener('click', () => {
        if (isEditorOpen) {
          closeEditor(false);
        } else {
          openEditor();
        }
      });

      // Set time action
      btnSetTime.addEventListener('click', () => {
        if (!isDraftValid()) return;

        confirmedTime = {
          hour12: draft.hour12,
          minute: draft.minute,
          ampm: draft.ampm
        };

        const hh = String(confirmedTime.hour12).padStart(2, '0');
        const mm = String(confirmedTime.minute).padStart(2, '0');
        const formatted = `${hh}:${mm} ${confirmedTime.ampm}`;

        slotDisplay.classList.remove('empty');
        slotDisplay.textContent = formatted;

        closeEditor(true);
      });

      // Cancel action (discards draft even if no prior committed time existed)
      btnCancelDraft.addEventListener('click', () => {
        // Discard uncommitted changes
        if (confirmedTime) {
          draft = {
            hour12: confirmedTime.hour12,
            minute: confirmedTime.minute,
            ampm: confirmedTime.ampm
          };
          rawTyped.hour = String(confirmedTime.hour12);
          rawTyped.minute = String(confirmedTime.minute).padStart(2, '0');
        } else {
          draft = { hour12: null, minute: null, ampm: null };
          rawTyped.hour = '';
          rawTyped.minute = '';
        }
        closeEditor(true);
      });

      // Unknown checkbox
      chkUnknown.addEventListener('change', (e) => {
        isUnknown = e.target.checked;
        clearOutcome();

        if (isUnknown) {
          if (isEditorOpen) {
            closeEditor(false);
          }
          timeSlotTrigger.classList.add('disabled');
          timeSlotTrigger.setAttribute('tabindex', '-1');
          timeSlotTrigger.setAttribute('aria-disabled', 'true');
          slotDisplay.classList.add('empty');
          slotDisplay.textContent = 'Time unknown';
          slotActionText.textContent = '—';
        } else {
          timeSlotTrigger.classList.remove('disabled');
          timeSlotTrigger.setAttribute('tabindex', '0');
          timeSlotTrigger.removeAttribute('aria-disabled');
          // Preserve retained known time when toggled off
          if (confirmedTime) {
            const hh = String(confirmedTime.hour12).padStart(2, '0');
            const mm = String(confirmedTime.minute).padStart(2, '0');
            slotDisplay.classList.remove('empty');
            slotDisplay.textContent = `${hh}:${mm} ${confirmedTime.ampm}`;
            slotActionText.textContent = 'Change →';
          } else {
            slotDisplay.classList.add('empty');
            slotDisplay.textContent = 'Tap to select time';
            slotActionText.textContent = 'Select →';
          }
        }
        updateBeginButtonState();
      });

      // Update Begin button enabled state
      function updateBeginButtonState() {
        if (isEditorOpen) {
          btnBegin.disabled = true;
          return;
        }
        if (isUnknown) {
          btnBegin.disabled = false;
        } else {
          btnBegin.disabled = confirmedTime === null;
        }
      }

      // Begin button click: Shows ONLY the actual entered time or 'Birth time unknown'
      btnBegin.addEventListener('click', () => {
        if (btnBegin.disabled) return;

        if (isUnknown) {
          outcomeDetailText.textContent = 'Birth time unknown';
        } else if (confirmedTime) {
          const hh = String(confirmedTime.hour12).padStart(2, '0');
          const mm = String(confirmedTime.minute).padStart(2, '0');
          outcomeDetailText.textContent = `${hh}:${mm} ${confirmedTime.ampm}`;
        }
        confirmedOutcomeCard.classList.add('visible');
        confirmedOutcomeCard.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
      });

      // Initial state
      updateDraftDisplay();
      updateBeginButtonState();
    })();
